ll must=0,tail=0;
vector<ll>dp(N);
bool dfs(ll vertex,ll par,vi &val){
    vis[vertex]=1;
    bool flag=true;
    for(auto &child:g[vertex]){
        if(vis[child] && child!=par){
            flag=false;
        }
        if(!vis[child]){
            flag&=dfs(child,vertex,val);
            dp[vertex]=max(dp[vertex],dp[child]);
        }

    }
    dp[vertex]+=val[vertex-1];
    if(flag){
        tail=max(tail,dp[vertex]);
    }
    else{
        must+=val[vertex-1];
    }
    return flag;
}
